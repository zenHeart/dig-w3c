下载代码示例

function downloadBlob(blob, filename) {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  // ⚠️ 关键：下载完成后立即释放
  URL.revokeObjectURL(url);
}

最佳实践：封装管理类

class ObjectURLManager {
  constructor() {
    this.urls = new Set();
  }

  create(blob) {
    const url = URL.createObjectURL(blob);
    this.urls.add(url);
    return url;
  }

  revoke(url) {
    if (this.urls.has(url)) {
      URL.revokeObjectURL(url);
      this.urls.delete(url);
    }
  }

  revokeAll() {
    this.urls.forEach(url => URL.revokeObjectURL(url));
    this.urls.clear();
  }
}

React 最佳实践

function ImagePreview({ file }) {
  const [url, setUrl] = useState(null);

  useEffect(() => {
    if (!file) return;
    const objectUrl = URL.createObjectURL(file);
    setUrl(objectUrl);
    // 清理函数：组件卸载时自动释放
    return () => URL.revokeObjectURL(objectUrl);
  }, [file]);

  return url ? <img src={url} /> : null;
}

Blob URL 生命周期

阶段状态说明
createObjectURL() ACTIVE URL 已创建，可正常使用
页面运行中 ACTIVE URL 有效，可用于 img.src 等
revokeObjectURL() REVOKED URL 已释放，不可再使用
标签页关闭 RELEASED 浏览器自动释放所有 Blob URL

自动释放场景

✅ 页面关闭/刷新：浏览器自动释放所有 Blob URL
✅ React useEffect 清理：组件卸载时调用 cleanup 函数
✅ img.src 覆盖：旧 URL 需要手动释放
⚠️ 频繁创建不释放：会导致内存泄漏

createObjectURL vs data URL vs FileReader

特性createObjectURLdata URLFileReader
返回类型 字符串 URL base64 字符串 Data URL / ArrayBuffer
性能 ✅ 快 ⚠️ 慢（编码开销） ⚠️ 异步
文件大小限制 无限制 URL 长度限制 无限制
内存管理 需手动释放 无需释放 无需释放
适用场景 图片预览、视频、下载 小图标、小文件 需要处理文件内容

使用建议

💡 大文件/媒体：使用 createObjectURL（性能最优）
💡 小图标/小文件：使用 data URL（内联方便）
💡 需要处理内容：使用 FileReader（读取文件数据）
